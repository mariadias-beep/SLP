13. Consolidação: condicionais e ciclos

Conteúdos para estudo

Combine as estruturas aprendidas. Para números aleatórios use rand() com srand(time(0)).

#include <cstdlib>
#include <ctime>
srand(time(0));
int secreto = rand() % 100 + 1;   // 1 a 100
int palpite, tent = 0;
do {
    cin >> palpite; tent++;
    if (palpite < secreto) cout << "Maior";
    else if (palpite > secreto) cout << "Menor";
} while (palpite != secreto);
